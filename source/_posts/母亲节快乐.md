<script>
(function () {
  const card = document.getElementById("momdayCard");
  if (!card) return;

  const photos = card.querySelectorAll(".momday-photo");
  const dots = card.querySelectorAll(".momday-dot");
  const prevButton = document.getElementById("momdayPrev");
  const nextButton = document.getElementById("momdayNext");
  const firstPhoto = document.getElementById("momdayFirstPhoto");
  const typingText = document.getElementById("momdayTyping");
  const pageOne = document.getElementById("momdayPageOne");
  const pageTwo = document.getElementById("momdayPageTwo");
  const playButton = document.getElementById("momdayPlayButton");
  const volume = document.getElementById("momdayVolume");
  const loveButton = document.getElementById("momdayLoveButton");
  const surprise = document.getElementById("momdaySurprise");
  const backButton = document.getElementById("momdayBackButton");
  const petals = document.getElementById("momdayPetals");

  let currentIndex = 0;
  let typedDone = false;
  let autoStarted = false;
  let audio = null;

  function goTo(index) {
    photos.forEach((photo) => photo.classList.remove("is-active"));
    dots.forEach((dot) => dot.classList.remove("is-active"));
    photos[index].classList.add("is-active");
    dots[index].classList.add("is-active");
    currentIndex = index;
  }

  function autoPlay() {
    window.setInterval(() => {
      goTo((currentIndex + 1) % photos.length);
    }, 3000);
  }

  function typeWrite() {
    typingText.style.display = "block";
    const text = typingText.innerText;
    typingText.innerText = "";
    let index = 0;

    function run() {
      if (index < text.length) {
        typingText.innerText += text[index];
        index += 1;
        window.setTimeout(run, 55);
      }
    }

    run();
  }

  function ensureAudio() {
    if (!audio) {
      audio = new Audio("/assets/music/%E9%9B%A8%E7%88%B1.mp3");
      audio.loop = true;
      audio.volume = volume.value;
    }
    return audio;
  }

  firstPhoto.addEventListener("click", () => {
    ensureAudio().play();
    playButton.innerText = "||";

    if (!typedDone) {
      typedDone = true;
      typeWrite();
      if (!autoStarted) {
        autoStarted = true;
        autoPlay();
      }
    }
  });

  playButton.addEventListener("click", () => {
    const player = ensureAudio();
    if (player.paused) {
      player.play();
      playButton.innerText = "||";
    } else {
      player.pause();
      playButton.innerText = "♪";
    }
  });

  volume.addEventListener("input", () => {
    if (audio) audio.volume = volume.value;
  });

  nextButton.addEventListener("click", () => {
    goTo((currentIndex + 1) % photos.length);
  });

  prevButton.addEventListener("click", () => {
    goTo((currentIndex - 1 + photos.length) % photos.length);
  });

  loveButton.addEventListener("click", () => {
    surprise.style.display = "block";
    loveButton.innerText = "爱您哟";
    loveButton.disabled = true;
    loveButton.style.background = "#ffccd5";
    loveButton.style.color = "#e63946";

    window.setTimeout(() => {
      pageOne.classList.add("momday-fly-away");
      window.setTimeout(() => {
        pageOne.style.display = "none";
        pageTwo.style.display = "block";
      }, 1200);
    }, 5000);
  });

  backButton.addEventListener("click", () => {
    pageTwo.style.display = "none";
    pageOne.style.display = "block";
    pageOne.classList.remove("momday-fly-away");
  });

  window.setInterval(() => {
    const petal = document.createElement("span");
    petal.className = "momday-petal";
    petal.style.left = Math.random() * 100 + "%";
    petal.style.animationDuration = Math.random() * 4 + 6 + "s";
    petals.appendChild(petal);
    window.setTimeout(() => petal.remove(), 10000);
  }, 400);
}());
</script>
